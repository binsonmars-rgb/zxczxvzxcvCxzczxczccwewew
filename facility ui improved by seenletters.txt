local BASE = "https://raw.githubusercontent.com/FacilityHUB/UI-Facility/refs/heads/main/"

local source = game:HttpGet(BASE .. "main.luau")
assert(#source > 1000 and not string.find(source, "404", 1, true),
	"main.luau introuvable a cette URL, verifie le depot et la branche")

local chunk, err = loadstring(source)
assert(chunk, "main.luau ne compile pas : " .. tostring(err))

local Library = chunk()
assert(type(Library) == "table", "main.luau n'a rien retourne, le fichier est incomplet ou tronque")

Library.BaseUrl = BASE

-- ============================================================
-- Inline SaveManager (doesn't depend on the repo addon)
-- ============================================================
local SaveManager = {}
local SM = {
	Library = Library,
	Folder = "Facility/game",
	Ignore = {},
	AutoloadFile = "autoload.txt",
}

local FS = Library.FileSystem

function SaveManager:SetLibrary(lib) SM.Library = lib end
function SaveManager:SetFolder(folder)
	SM.Folder = folder
	Library:BuildFolders(folder)
end
function SaveManager:SetIgnoreIndexes(list) SM.Ignore = list or {} end
function SaveManager:IgnoreThemeSettings() end

function SaveManager:GetConfigPath(name)
	return SM.Folder .. "/configs/" .. tostring(name) .. ".json"
end

function SaveManager:GetAutoloadPath()
	return SM.Folder .. "/settings/" .. SM.AutoloadFile
end

function SaveManager:Save(name)
	if not name or name == "" then return false end
	local data = SM.Library:GetConfig(SM.Ignore)
	local ok, encoded = pcall(function()
		return game:GetService("HttpService"):JSONEncode(data)
	end)
	if not ok then return false end
	Library:BuildFolders(SM.Folder)
	Library:BuildFolders(SM.Folder .. "/configs")
	FS:Write(self:GetConfigPath(name), encoded)
	return true
end

function SaveManager:Load(name)
	if not name or name == "" then return false end
	local raw = FS:Read(self:GetConfigPath(name))
	if not raw then return false end
	local ok, data = pcall(function()
		return game:GetService("HttpService"):JSONDecode(raw)
	end)
	if not ok or type(data) ~= "table" then return false end
	SM.Library:LoadConfig(data, SM.Ignore)
	return true
end

function SaveManager:Delete(name)
	if not name or name == "" then return false end
	FS:Delete(self:GetConfigPath(name))
	return true
end

function SaveManager:List()
	local files = FS:List(SM.Folder .. "/configs")
	local out = {}
	for _, f in ipairs(files) do
		local name = string.match(f, "^(.+)%.json$")
		if name then table.insert(out, name) end
	end
	table.sort(out)
	return out
end

function SaveManager:RefreshConfigList() end
function SaveManager:GetConfigList() return self:List() end

function SaveManager:SetAutoloadConfig(name)
	Library:BuildFolders(SM.Folder)
	Library:BuildFolders(SM.Folder .. "/settings")
	FS:Write(self:GetAutoloadPath(), tostring(name or ""))
	return true
end

function SaveManager:GetAutoload()
	local raw = FS:Read(self:GetAutoloadPath())
	if not raw or raw == "" then return nil end
	return raw
end

function SaveManager:ClearAutoload() return self:SetAutoloadConfig("") end

function SaveManager:LoadAutoloadConfig()
	local name = self:GetAutoload()
	if not name or name == "" then return false end
	return self:Load(name)
end

Library.SaveManager = SaveManager
